from pathlib import Path
import pandas as pd
import matplotlib.pyplot as plt

from pcntoolkit import (
    BLR,
    NormativeModel,
    NormData,
    plot_centiles,
    plot_qq
)

import pcntoolkit.util.output


# ============================================================
# SETUP
# ============================================================

current_dir = Path.cwd().parent

model_df = pd.read_csv(
    current_dir
    / "results"
    / "combined_dataset"
    / "Model_dataframe.csv"
)

pcntoolkit.util.output.Output.set_show_messages(False)


# ============================================================
# DEFINE MODEL VARIABLES
# ============================================================

response_vars = [f"comp{i}" for i in range(1, 19)]

batch_effects = [
    "Dataset_name"
]

covariates = [
    col for col in model_df.columns
    if col not in (
        ["Scan_id", "Participant_Id", "Dataset_name"]
        + response_vars
    )
]


# ============================================================
# CREATE NORMDATA
# ============================================================

norm_data = NormData.from_dataframe(
    name="DigitalRodent",
    dataframe=model_df,
    covariates=covariates,
    batch_effects=batch_effects,
    response_vars=response_vars,
    remove_Nan=False
)


# ============================================================
# TRAIN / TEST SPLIT
# ============================================================

train, test = norm_data.train_test_split()


# ============================================================
# BLR MODEL
# ============================================================

model = NormativeModel(
    BLR(
        heteroskedastic=True
    ),
    inscaler="standardize",
    outscaler="standardize",
    savemodel=True,
    evaluate_model=True,
    saveresults=True,
    saveplots=True,
    save_dir=str(
        current_dir
        / "results"
        / "normative_model"
        / "BLR"
    )
)


# ============================================================
# FIT MODEL
# ============================================================

model.fit_predict(
    train,
    test
)


# ============================================================
# PLOT DIRECTORY
# ============================================================

plot_dir = (
    current_dir
    / "results"
    / "normative_model"
    / "BLR"
    / "plots"
)

plot_dir.mkdir(
    parents=True,
    exist_ok=True
)


# ============================================================
# CENTILE PLOTS
# ============================================================

continuous_covariates = [
    "Age(week)",
    "MRI",
    "MRI T.R.",
    "FD average",
    "FD maximum"
]

# Only keep covariates that are actually present
continuous_covariates = [
    col for col in continuous_covariates
    if col in covariates
]

for covariate in continuous_covariates:

    plt.figure(figsize=(6, 4))

    plot_centiles(
        model,
        scatter_data=train,
        centiles=[0.05, 0.5, 0.95],
        covariate=covariate,
        scatter_kwargs={
            "s": 10,
            "alpha": 0.4
        }
    )

    plt.title(f"Centile plot: {covariate}")
    plt.tight_layout()

    safe_name = (
        covariate
        .replace(" ", "_")
        .replace(".", "_")
        .replace("/", "_")
        .replace("(", "")
        .replace(")", "")
    )

    plt.savefig(
        plot_dir / f"centiles_{safe_name}.png",
        dpi=300,
        bbox_inches="tight"
    )

    plt.show()
    plt.close()


# ============================================================
# QQ PLOTS
# ============================================================

qq_dir = (
    current_dir
    / "results"
    / "normative_model"
    / "BLR"
    / "plots"
    / "qq"
)

qq_dir.mkdir(
    parents=True,
    exist_ok=True
)

plot_qq(
    test,
    plot_id_line=True,
    save_dir=str(qq_dir)
)
